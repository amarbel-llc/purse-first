# mgp (model graph protocol) Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build a new MCP package called mgp that exposes the entire purse-first tool ecosystem through two MCP tools: `query` (GraphQL) and `exec` (proxy to underlying MCP servers).

**Architecture:** mgp is an MCP server built with `command.App` from `libs/go-mcp`. At startup it discovers installed MCP packages via marketplace plugin.json files, introspects each by spawning the binary and calling `tools/list`, then builds a GraphQL schema from the collected tool metadata. It serves two MCP tools over stdio.

**Tech Stack:** Go, `libs/go-mcp` (command framework), `github.com/graphql-go/graphql` (GraphQL engine), `libs/go-mcp/jsonrpc` + `libs/go-mcp/transport` (MCP client over stdio)

```go
		if err := app.GenerateAll(flag.Arg(1)); err != nil {
			log.Fatalf("generating plugin: %v", err)
		}
		return
	}

	if flag.NArg() > 0 {
		fmt.Fprintf(os.Stderr, "mgp: unexpected arguments: %v\n", flag.Args())
		flag.Usage()
		os.Exit(1)
	}

	defer cancel()

		ServerVersion: app.Version,
		Instructions:  "Model graph protocol — query and execute purse-first MCP tools via GraphQL.",
		Tools:         registry,
	})
	if err != nil {
		log.Fatalf("creating server: %v", err)
	}

	if err := srv.Run(ctx); err != nil {
		log.Fatalf("server error: %v", err)
	}
```

```go
package catalog

import "encoding/json"

// CatalogTool holds metadata for a single MCP tool discovered from an MCP server.
type CatalogTool struct {
	Name        string          `json:"name"`
	Title       string          `json:"title,omitempty"`
	Description string          `json:"description,omitempty"`
	Package     string          `json:"package"`
	InputSchema json.RawMessage `json:"inputSchema,omitempty"`
	ReadOnly    *bool           `json:"readOnly,omitempty"`
	Destructive *bool           `json:"destructive,omitempty"`
	Idempotent  *bool           `json:"idempotent,omitempty"`
	OpenWorld   *bool           `json:"openWorld,omitempty"`
}

// ServerEntry holds the command info needed to spawn an MCP server.
type ServerEntry struct {
	Name    string
	Command string
	Args    []string
}

// Catalog holds the complete tool catalog for all discovered MCP servers.
type Catalog struct {
	Tools   []CatalogTool
	Servers map[string]ServerEntry // keyed by server name
}

// NewCatalog creates an empty catalog.
func NewCatalog() *Catalog {
	return &Catalog{
		Servers: make(map[string]ServerEntry),
	}
}

// AddTool adds a tool to the catalog.
func (c *Catalog) AddTool(tool CatalogTool) {
	c.Tools = append(c.Tools, tool)
}

// AddServer registers an MCP server entry for exec lookups.
func (c *Catalog) AddServer(entry ServerEntry) {
	c.Servers[entry.Name] = entry
}

// FindServer looks up a server by name.
func (c *Catalog) FindServer(name string) (ServerEntry, bool) {
	s, ok := c.Servers[name]
	return s, ok
}
```

### Task 3: Implement the minimal MCP client

**Files:**
- Create: `packages/mgp/internal/mcpclient/client.go`

The client speaks newline-delimited JSON-RPC over stdio to child MCP processes. It needs to do three things: `initialize`, `tools/list`, and `tools/call`.

**Step 1: Write the MCP client**

```go
package mcpclient

import (
	"context"
	"encoding/json"
	"fmt"
	"io"
	"os/exec"
	"sync/atomic"

	"github.com/amarbel-llc/purse-first/libs/go-mcp/jsonrpc"
	"github.com/amarbel-llc/purse-first/libs/go-mcp/protocol"
	"github.com/amarbel-llc/purse-first/libs/go-mcp/transport"
)

// Client is a minimal MCP client that communicates over stdio with a child process.
type Client struct {
	cmd   *exec.Cmd
	stdio *transport.Stdio
	nextID atomic.Int64
	stdin io.WriteCloser
}

// Spawn starts an MCP server process and returns a connected client.
func Spawn(ctx context.Context, command string, args ...string) (*Client, error) {
	cmd := exec.CommandContext(ctx, command, args...)
	stdin, err := cmd.StdinPipe()
	if err != nil {
		return nil, fmt.Errorf("creating stdin pipe: %w", err)
	}
	stdout, err := cmd.StdoutPipe()
	if err != nil {
		stdin.Close()
		return nil, fmt.Errorf("creating stdout pipe: %w", err)
	}
	cmd.Stderr = nil // discard stderr

	if err := cmd.Start(); err != nil {
		stdin.Close()
		return nil, fmt.Errorf("starting process: %w", err)
	}

	return &Client{
		cmd:   cmd,
		stdio: transport.NewStdio(stdout, stdin),
		stdin: stdin,
	}, nil
}

func (c *Client) nextRequestID() jsonrpc.ID {
	return jsonrpc.NewNumberID(c.nextID.Add(1))
}

// call sends a JSON-RPC request and waits for the response.
func (c *Client) call(method string, params any) (json.RawMessage, error) {
	req, err := jsonrpc.NewRequest(c.nextRequestID(), method, params)
	if err != nil {
		return nil, fmt.Errorf("creating request: %w", err)
	}

	if err := c.stdio.Write(req); err != nil {
		return nil, fmt.Errorf("writing request: %w", err)
	}

	// Read responses, skipping notifications until we get a response.
	for {
		msg, err := c.stdio.Read()
		if err != nil {
			return nil, fmt.Errorf("reading response: %w", err)
		}

		if msg.IsNotification() {
			continue
		}

		if msg.Error != nil {
			return nil, msg.Error
		}

		return msg.Result, nil
	}
}

// Initialize performs the MCP initialize handshake.
func (c *Client) Initialize(ctx context.Context) error {
	params := protocol.InitializeParamsV1{
		ProtocolVersion: "2025-11-25",
		ClientInfo: protocol.ImplementationV1{
			Name:    "mgp",
			Version: "0.1.0",
		},
		Capabilities: protocol.ClientCapabilitiesV1{},
	}

	if _, err := c.call("initialize", params); err != nil {
		return fmt.Errorf("initialize: %w", err)
	}

	// Send initialized notification
	notif, err := jsonrpc.NewNotification("notifications/initialized", nil)
	if err != nil {
		return fmt.Errorf("creating initialized notification: %w", err)
	}

	return c.stdio.Write(notif)
}

// ListTools calls tools/list and returns the V1 tool definitions.
func (c *Client) ListTools(ctx context.Context) ([]protocol.ToolV1, error) {
	result, err := c.call("tools/list", map[string]any{})
	if err != nil {
		return nil, fmt.Errorf("tools/list: %w", err)
	}

	var toolsResult protocol.ToolsListResultV1
	if err := json.Unmarshal(result, &toolsResult); err != nil {
		return nil, fmt.Errorf("parsing tools/list result: %w", err)
	}

	return toolsResult.Tools, nil
}

// CallTool calls tools/call and returns the raw result.
func (c *Client) CallTool(ctx context.Context, name string, args json.RawMessage) (json.RawMessage, error) {
	params := map[string]any{
		"name": name,
	}
	if args != nil {
		var argsMap map[string]any
		if err := json.Unmarshal(args, &argsMap); err == nil {
			params["arguments"] = argsMap
		}
	}

	return c.call("tools/call", params)
}

// Close kills the child process and cleans up.
func (c *Client) Close() error {
	c.stdin.Close()
	return c.cmd.Wait()
}
```

### Task 4: Implement catalog population from marketplace discovery

**Files:**
- Create: `packages/mgp/internal/catalog/discover.go`

This reads plugin.json files from a plugins directory (the share/purse-first/ tree), then spawns each MCP server to introspect its tools via `tools/list`.

**Step 1: Write the discovery + introspection logic**

```go
package catalog

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"os"
	"path/filepath"
	"time"

	"github.com/amarbel-llc/mgp/internal/mcpclient"
	"github.com/amarbel-llc/purse-first/libs/go-mcp/protocol"
)

const introspectTimeout = 10 * time.Second

type pluginManifest struct {
	Name       string                       `json:"name"`
	McpServers map[string]mcpServerManifest `json:"mcpServers,omitempty"`
}

type mcpServerManifest struct {
	Type    string   `json:"type"`
	Command string   `json:"command"`
	Args    []string `json:"args,omitempty"`
}

// Discover reads plugin.json files from pluginsDir and introspects each MCP
// server to populate the catalog. selfName is excluded from discovery.
func Discover(ctx context.Context, pluginsDir, binDir, selfName string) (*Catalog, error) {
	cat := NewCatalog()

	matches, err := filepath.Glob(filepath.Join(pluginsDir, "*", "plugin.json"))
	if err != nil {
		return cat, fmt.Errorf("globbing plugin manifests: %w", err)
	}

	for _, path := range matches {
		data, err := os.ReadFile(path)
		if err != nil {
			continue
		}

		var manifest pluginManifest
		if err := json.Unmarshal(data, &manifest); err != nil {
			continue
		}

		pluginName := filepath.Base(filepath.Dir(path))
		if pluginName == selfName {
			continue
		}

		for serverName, srv := range manifest.McpServers {
			if srv.Type != "" && srv.Type != "stdio" {
				continue
			}

			command := srv.Command
			if binDir != "" {
				command = filepath.Join(binDir, srv.Command)
			}

			entry := ServerEntry{
				Name:    serverName,
				Command: command,
				Args:    srv.Args,
			}
			cat.AddServer(entry)

			tools, err := introspectServer(ctx, entry)
			if err != nil {
				log.Printf("warning: failed to introspect %s: %v", serverName, err)
				continue
			}

			for _, tool := range tools {
				ct := CatalogTool{
					Name:        tool.Name,
					Title:       tool.Title,
					Description: tool.Description,
					Package:     serverName,
					InputSchema: tool.InputSchema,
				}
				if tool.Annotations != nil {
					ct.ReadOnly = tool.Annotations.ReadOnlyHint
					ct.Destructive = tool.Annotations.DestructiveHint
					ct.Idempotent = tool.Annotations.IdempotentHint
					ct.OpenWorld = tool.Annotations.OpenWorldHint
				}
				cat.AddTool(ct)
			}
		}
	}

	return cat, nil
}

func introspectServer(ctx context.Context, entry ServerEntry) ([]protocol.ToolV1, error) {
	ctx, cancel := context.WithTimeout(ctx, introspectTimeout)
	defer cancel()

	client, err := mcpclient.Spawn(ctx, entry.Command, entry.Args...)
	if err != nil {
		return nil, fmt.Errorf("spawning %s: %w", entry.Name, err)
	}
	defer client.Close()

	if err := client.Initialize(ctx); err != nil {
		return nil, fmt.Errorf("initializing %s: %w", entry.Name, err)
	}

	tools, err := client.ListTools(ctx)
	if err != nil {
		return nil, fmt.Errorf("listing tools for %s: %w", entry.Name, err)
	}

	return tools, nil
}
```

```go
package graphqlschema

import (
	"encoding/json"
	"strings"

	"github.com/amarbel-llc/mgp/internal/catalog"
	"github.com/graphql-go/graphql"
)

func BuildSchema(cat *catalog.Catalog) (graphql.Schema, error) {
	toolType := graphql.NewObject(graphql.ObjectConfig{
		Name: "Tool",
		Fields: graphql.Fields{
			"name":        &graphql.Field{Type: graphql.NewNonNull(graphql.String)},
			"title":       &graphql.Field{Type: graphql.String},
			"description": &graphql.Field{Type: graphql.String},
			"package":     &graphql.Field{Type: graphql.NewNonNull(graphql.String)},
			"inputSchema": &graphql.Field{
				Type: graphql.String,
				Resolve: func(p graphql.ResolveParams) (any, error) {
					tool := p.Source.(catalog.CatalogTool)
					if tool.InputSchema == nil {
						return nil, nil
					}
					return string(tool.InputSchema), nil
				},
			},
			"readOnly": &graphql.Field{
				Type: graphql.Boolean,
				Resolve: func(p graphql.ResolveParams) (any, error) {
					tool := p.Source.(catalog.CatalogTool)
					if tool.ReadOnly == nil {
						return nil, nil
					}
					return *tool.ReadOnly, nil
				},
			},
			"destructive": &graphql.Field{
				Type: graphql.Boolean,
				Resolve: func(p graphql.ResolveParams) (any, error) {
					tool := p.Source.(catalog.CatalogTool)
					if tool.Destructive == nil {
						return nil, nil
					}
					return *tool.Destructive, nil
				},
			},
			"idempotent": &graphql.Field{
				Type: graphql.Boolean,
				Resolve: func(p graphql.ResolveParams) (any, error) {
					tool := p.Source.(catalog.CatalogTool)
					if tool.Idempotent == nil {
						return nil, nil
					}
					return *tool.Idempotent, nil
				},
			},
			"openWorld": &graphql.Field{
				Type: graphql.Boolean,
				Resolve: func(p graphql.ResolveParams) (any, error) {
					tool := p.Source.(catalog.CatalogTool)
					if tool.OpenWorld == nil {
						return nil, nil
					}
					return *tool.OpenWorld, nil
				},
			},
		},
	})

	queryType := graphql.NewObject(graphql.ObjectConfig{
		Name: "Query",
		Fields: graphql.Fields{
			"tools": &graphql.Field{
				Type: graphql.NewNonNull(graphql.NewList(graphql.NewNonNull(toolType))),
				Args: graphql.FieldConfigArgument{
					"package": &graphql.ArgumentConfig{
						Type: graphql.String,
					},
					"name": &graphql.ArgumentConfig{
						Type: graphql.String,
					},
					"readOnly": &graphql.ArgumentConfig{
						Type: graphql.Boolean,
					},
					"destructive": &graphql.ArgumentConfig{
						Type: graphql.Boolean,
					},
				},
				Resolve: func(p graphql.ResolveParams) (any, error) {
					var result []catalog.CatalogTool

					pkgFilter, _ := p.Args["package"].(string)
					nameFilter, _ := p.Args["name"].(string)
					readOnlyFilter, hasReadOnly := p.Args["readOnly"].(bool)
					destructiveFilter, hasDestructive := p.Args["destructive"].(bool)

					for _, tool := range cat.Tools {
						if pkgFilter != "" && tool.Package != pkgFilter {
							continue
						}
						if nameFilter != "" && !strings.Contains(tool.Name, nameFilter) {
							continue
						}
						if hasReadOnly && (tool.ReadOnly == nil || *tool.ReadOnly != readOnlyFilter) {
							continue
						}
						if hasDestructive && (tool.Destructive == nil || *tool.Destructive != destructiveFilter) {
							continue
						}
						result = append(result, tool)
					}

					if result == nil {
						result = []catalog.CatalogTool{}
					}

					return result, nil
				},
			},
		},
	})

	return graphql.NewSchema(graphql.SchemaConfig{
		Query: queryType,
	})
}

// Execute runs a GraphQL query against the schema and returns JSON.
func Execute(schema graphql.Schema, query string) (json.RawMessage, error) {
	result := graphql.Do(graphql.Params{
		Schema:        schema,
		RequestString: query,
	})

	return json.Marshal(result)
}
```

```go
package tools

import (
	"context"
	"encoding/json"
	"fmt"

	"github.com/amarbel-llc/mgp/internal/catalog"
	"github.com/amarbel-llc/mgp/internal/mcpclient"
	"github.com/amarbel-llc/purse-first/libs/go-mcp/command"
	"github.com/amarbel-llc/purse-first/libs/go-mcp/protocol"
)

func registerExecCommand(app *command.App, cat *catalog.Catalog) {
	app.AddCommand(&command.Command{
		Name:        "exec",
		Title:       "Execute MCP Tool",
		Description: command.Description{Short: "Execute a tool on an MCP server"},
		Annotations: &protocol.ToolAnnotations{
			ReadOnlyHint: protocol.BoolPtr(false),
		},
		Params: []command.Param{
			{Name: "server", Type: command.String, Description: "MCP server name (e.g. grit, get-hubbed, chix)", Required: true},
			{Name: "tool", Type: command.String, Description: "Tool name to call", Required: true},
			{Name: "args", Type: command.String, Description: "Arguments to pass to the tool as JSON object"},
		},
		Run: func(ctx context.Context, rawArgs json.RawMessage, _ command.Prompter) (*command.Result, error) {
			var params struct {
				Server string          `json:"server"`
				Tool   string          `json:"tool"`
				Args   json.RawMessage `json:"args"`
			}
			if err := json.Unmarshal(rawArgs, &params); err != nil {
				return command.TextErrorResult(fmt.Sprintf("invalid arguments: %v", err)), nil
			}

			entry, ok := cat.FindServer(params.Server)
			if !ok {
				return command.TextErrorResult(fmt.Sprintf("unknown server: %s", params.Server)), nil
			}

			client, err := mcpclient.Spawn(ctx, entry.Command, entry.Args...)
			if err != nil {
				return command.TextErrorResult(fmt.Sprintf("failed to spawn %s: %v", params.Server, err)), nil
			}
			defer client.Close()

			if err := client.Initialize(ctx); err != nil {
				return command.TextErrorResult(fmt.Sprintf("failed to initialize %s: %v", params.Server, err)), nil
			}

			result, err := client.CallTool(ctx, params.Tool, params.Args)
			if err != nil {
				return command.TextErrorResult(fmt.Sprintf("tool call failed: %v", err)), nil
			}

			return &command.Result{
```

### Task 7: Wire up main.go with catalog discovery

**Files:**
- Modify: `packages/mgp/cmd/mgp/main.go`

**Step 1: Update main.go to discover catalog at startup**

Replace the main.go content to integrate catalog discovery before registering tools. The key change is: discover the catalog before calling `tools.RegisterAll`, and pass the catalog in.

The updated main.go needs to:
1. Accept a `--plugins-dir` flag for the share/purse-first/ directory
2. Accept a `--bin-dir` flag for the bin/ directory where MCP binaries live
3. Call `catalog.Discover` to populate the catalog
4. Pass the catalog to `tools.RegisterAll`

```go
package main

import (
	"context"
	"flag"
	"fmt"
	"log"
	"os"
	"os/signal"
	"path/filepath"

	"github.com/amarbel-llc/mgp/internal/catalog"
	"github.com/amarbel-llc/mgp/internal/tools"
	"github.com/amarbel-llc/purse-first/libs/go-mcp/server"
	"github.com/amarbel-llc/purse-first/libs/go-mcp/transport"
)

func main() {
	pluginsDir := flag.String("plugins-dir", "", "Path to share/purse-first/ directory containing plugin.json files")
	binDir := flag.String("bin-dir", "", "Path to bin/ directory containing MCP server binaries")

	flag.Usage = func() {
		fmt.Fprintf(os.Stderr, "mgp — model graph protocol MCP server\n\n")
		fmt.Fprintf(os.Stderr, "Usage:\n")
		fmt.Fprintf(os.Stderr, "  mgp [flags]\n\n")
		fmt.Fprintf(os.Stderr, "Exposes the purse-first tool catalog via GraphQL.\n\n")
		fmt.Fprintf(os.Stderr, "Flags:\n")
		flag.PrintDefaults()
	}

	flag.Parse()

	// For generate-plugin, we don't need a catalog
	if flag.NArg() == 2 && flag.Arg(0) == "generate-plugin" {
		app := tools.RegisterAll(catalog.NewCatalog())
		if err := app.GenerateAll(flag.Arg(1)); err != nil {
			log.Fatalf("generating plugin: %v", err)
		}
		return
	}

	if flag.NArg() > 0 {
		fmt.Fprintf(os.Stderr, "mgp: unexpected arguments: %v\n", flag.Args())
		flag.Usage()
		os.Exit(1)
	}

	ctx, cancel := signal.NotifyContext(context.Background(), os.Interrupt)
	defer cancel()

	// Discover catalog
	pd := *pluginsDir
	bd := *binDir
	if pd == "" {
		// Default: look relative to own executable
		exe, err := os.Executable()
		if err == nil {
			base := filepath.Dir(filepath.Dir(exe)) // go from bin/ to root
			pd = filepath.Join(base, "share", "purse-first")
			if bd == "" {
				bd = filepath.Join(base, "bin")
			}
		}
	}

	cat, err := catalog.Discover(ctx, pd, bd, "mgp")
	if err != nil {
		log.Printf("warning: catalog discovery failed: %v", err)
		cat = catalog.NewCatalog()
	}

	app := tools.RegisterAll(cat)

	t := transport.NewStdio(os.Stdin, os.Stdout)

	registry := server.NewToolRegistryV1()
	app.RegisterMCPToolsV1(registry)

	srv, err := server.New(t, server.Options{
		ServerName:    app.Name,
		ServerVersion: app.Version,
		Instructions:  "Model graph protocol — query and execute purse-first MCP tools via GraphQL.",
		Tools:         registry,
	})
	if err != nil {
		log.Fatalf("creating server: %v", err)
	}

	if err := srv.Run(ctx); err != nil {
		log.Fatalf("server error: %v", err)
	}
}
```

Run: `nix develop --command go build ./packages/mgp/cmd/mgp`
